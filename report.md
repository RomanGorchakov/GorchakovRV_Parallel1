## Теоретическая часть
### Изученные концепции
- **MPI (Message Passing Interface):** стандарт для организации параллельных вычислений в системах с распределенной памятью. В таких системах каждый процессор имеет свою локальную память, и для обмена данными между процессами используется передача сообщений.

### Ключевые термины
- **Ранг:** уникальный номер процесса в коммуникаторе
- **Коммуникатор:** группа процессов, участвующих в вычислениях
- **Операции точка-точка (Send и Recv):** передача данных между двумя процессами
- **Коллективные операции (Bcast, Scatterv, Gatherv):** — взаимодействие всех процессов

```python
#!/usr/bin/env python3
import numpy as np
import time

def read_params(filename="data/in.dat"):
    """Чтение размеров матрицы из файла"""
    f1 = open('data/in.dat', 'r')
    N = int(f1.readline())
    M = int(f1.readline())
    f1.close()
    
    return N, M

def read_matrix(filename="data/AData.dat", M=None, N=None):
    """Чтение матрицы из файла"""
    A = np.empty((M, N))
    x = np.empty(N)
    b = np.empty(M)
    
    f2 = open('data/AData.dat', 'r')
    for j in range(M):
        for i in range(N):
            A[j, i] = float(f2.readline())
    f2.close()
    
    return A

def read_vector(filename="data/xData.dat"):
    """Чтение вектора из файла"""
    f3 = open('data/xData.dat', 'r')
    x = np.array([float(line) for line in f3 if line.strip()])
    f3.close()
    
    return x

def sequential_mat_vec_mult(A, x):
    """Последовательное умножение матрицы на вектор"""
    M, N = A.shape
    b = np.empty(M)
    for j in range(M):
        b[j] = 0
        for i in range(N):
            b[j] += A[j, i] * x[i]
            
    return b

def main():
    """Основная функция"""
    N, M = read_params(filename="data/in.dat")
    A = read_matrix(filename="data/AData.dat", M=M, N=N)
    x = read_vector(filename="data/xData.dat")
    
    start_time = time.time()
    b = sequential_mat_vec_mult(A, x)
    elapsed_time = time.time() - start_time
    
    print(f"Execution time: {elapsed_time:.4f} seconds")
    
    f4 = open('Results_sequential.dat', 'w')
    for j in range(M):
        print(b[j], file=f4)
    f4.close()

if __name__ == "__main__":
    main()
```

#### Задача 2: Параллельная версия с Send/Recv
**Цель:** Реализовать параллельную версию с использованием операций точка-точка Send и Recv.

**Код:**
```python
#!/usr/bin/env python3
from mpi4py import MPI
import numpy as np
import time

def read_params(filename="data/in.dat"):
    # Чтение параметров
    f1 = open('data/in.dat', 'r')
    N = int(f1.readline().strip())
    M = int(f1.readline().strip())
    f1.close()
    
    return N, M

def auxiliary_arrays_determination(M, numprocs):
    """Расчет rcounts и displs для разбиения данных"""
    num_workers = numprocs - 1 if numprocs > 1 else 1
    base = M // num_workers
    rem = M % num_workers

    rcounts = np.array([base + (1 if i < rem else 0) for i in range(num_workers)], dtype=np.int32)
    displs = np.array([sum(rcounts[:i]) for i in range(num_workers)], dtype=np.int32)

    return rcounts, displs

def main():
    # Инициализация MPI
    comm = MPI.COMM_WORLD
    rank = comm.Get_rank()
    numprocs = comm.Get_size()
    
    start_time = None
    
    # Чтение параметров на процессе 0
    if rank == 0:
        start_time = time.time()
        N_val, M_val = read_params("data/in.dat")
        params = np.array([N_val, M_val], dtype=np.int32)
    else:
        params = np.empty(2, dtype=np.int32)
    
    # Рассылка параметров через Bcast
    comm.Bcast([params, 2, MPI.INT], root=0)
    N, M = params[0], params[1]
    
    # Определение схемы распределения нагрузки
    num_workers = numprocs - 1 if numprocs > 1 else 1
    rcounts, displs = auxiliary_arrays_determination(M, numprocs)
    
    # Распределение матрицы через Send/Recv
    if rank == 0:
        f2 = open("data/AData.dat", "r")

        if numprocs > 1:
            # Чтение и отправление подматрицы для каждого рабочего процесса
            for k in range(1, numprocs):
                local_m = rcounts[k - 1]
                A_part = np.empty((local_m, N), dtype=np.float64)
                for j in range(local_m):
                    for i in range(N):
                        A_part[j, i] = float(f2.readline())
                comm.Send([A_part, MPI.DOUBLE], dest=k, tag=0)
            f2.close()
            A_part = None
        else:
            A_part = np.empty((M, N), dtype=np.float64)
            for j in range(M):
                for i in range(N):
                    A_part[j, i] = float(f2.readline())
            f2.close()

    else:
        # Рабочие процессы получают свою часть матрицы A
        local_m = rcounts[rank - 1]
        A_part = np.empty((local_m, N), dtype=np.float64)
        comm.Recv([A_part, MPI.DOUBLE], source=0, tag=0)
    
    # Распределение вектора через Bcast
    x = np.empty(N, dtype=np.float64)
    
    if rank == 0:
        f3 = open('data/xData.dat', 'r')
        for i in range(N):
            x[i] = float(f3.readline())
        f3.close()

    comm.Bcast([x, MPI.DOUBLE], root=0)
    
    # Локальное вычисление
    if numprocs > 1:
        if rank != 0:
            b_part = np.dot(A_part, x)
    else:
        b_part = np.dot(A_part, x)
        
    # Сбор результатов через Send/Recv
    if rank == 0:
        b = np.empty(M, dtype=np.float64)

        if numprocs > 1:
            for k in range(1, numprocs):
                local_m = rcounts[k - 1]
                offset = displs[k - 1]
                comm.Recv([b[offset : offset + local_m], MPI.DOUBLE], source=k, tag=0)
        else:
            b = b_part

    else:
        comm.Send([b_part, MPI.DOUBLE], dest=0, tag=0)
        
    # Запись результата
    if rank == 0:
        elapsed_time = time.time() - start_time
        print(f"Execution time: {elapsed_time:.4f} seconds")

        f4 = open('Results_parallel_send_recv.dat', 'w')
        for j in range(M):
            print(b[j], file=f4)
        f4.close()

if __name__ == "__main__":
    main()
```

#### Задача 3: Параллельная версия с Scatterv/Gatherv
**Цель:** Реализовать параллельную версию с использованием коллективных операций Scatterv и Gatherv.

**Код:**
```python
#!/usr/bin/env python3
from mpi4py import MPI
import numpy as np
import time

def read_params(filename="data/in.dat"):
    # Чтение параметров
    f1 = open('data/in.dat', 'r')
    N = int(f1.readline().strip())
    M = int(f1.readline().strip())
    f1.close()
    
    return np.array(N, dtype=np.int32), np.array(M, dtype=np.int32)

def auxiliary_arrays_determination(M, numprocs):
    """Расчет rcounts и displs для разбиения данных"""
    rcounts = np.zeros(numprocs, dtype=np.int32)
    displs = np.zeros(numprocs, dtype=np.int32)

    if numprocs > 1:
        ave, res = divmod(M, numprocs - 1)
        # Процесс 0 не участвует в вычислениях
        rcounts[0] = 0
        displs[0] = 0
        for k in range(1, numprocs):
            if k <= res:
                rcounts[k] = ave + 1
            else:
                rcounts[k] = ave
            displs[k] = displs[k-1] + rcounts[k-1]
    else:
        # Если запущен всего 1 процесс, он делает всю работу
        rcounts[0] = M
        displs[0] = 0

    return rcounts, displs

def main():
    # Инициализация MPI
    comm = MPI.COMM_WORLD
    rank = comm.Get_rank()
    numprocs = comm.Get_size()
    
    # Чтение параметров на процессе 0
    N = np.array(0, dtype=np.int32)
    M = np.array(0, dtype=np.int32)

    if rank == 0:
        N, M = read_params("data/in.dat")

    # Рассылка параметров через Bcast
    comm.Bcast([N, 1, MPI.INT], root=0)

    # Определение вспомогательных массивов для Scatterv на процессе 0
    if rank == 0:
        rcounts, displs = auxiliary_arrays_determination(int(M), numprocs)
        sendbuf_M = [rcounts, np.ones(numprocs, dtype=np.int32), np.arange(numprocs, dtype=np.int32), MPI.INT]
    else:
        rcounts = None
        displs = None
        sendbuf_M = None

    # Распределение количества строк для каждого процесса
    M_part = np.array(0, dtype=np.int32)
    comm.Scatterv(sendbuf_M, [M_part, 1, MPI.INT], root=0)

    # Выделение памяти под локальную часть матрицы A
    A_part = np.empty((int(M_part), int(N)), dtype=np.float64)
    
    # Распределение матрицы через Scatterv
    if rank == 0:
        # Чтение матрицы A на процессе 0
        f2 = open('data/AData.dat', 'r')
        A = np.empty((int(M), int(N)), dtype=np.float64)
        for j in range(int(M)):
            for i in range(int(N)):
                A[j, i] = np.float64(f2.readline())
        f2.close()

        sendbuf_A = [A, rcounts * int(N), displs * int(N), MPI.DOUBLE]
    else:
        sendbuf_A = None

    comm.Scatterv(sendbuf_A, [A_part, int(M_part * N), MPI.DOUBLE], root=0)
    
    # Распределение вектора через Bcast
    x = np.empty(int(N), dtype=np.float64)
    if rank == 0:
        f3 = open('data/xData.dat', 'r')
        for i in range(int(N)):
            x[i] = np.float64(f3.readline())
        f3.close()

    comm.Bcast([x, int(N), MPI.DOUBLE], root=0)

    # Замер времени начала вычислений
    comm.Barrier()
    start_time = time.time()
    
    # Локальное вычисление
    b_part = np.dot(A_part, x)

    comm.Barrier()
    
    # Сбор результатов через Gatherv
    if rank == 0:
        b = np.empty(int(M), dtype=np.float64)
        recvbuf_b = [b, rcounts, displs, MPI.DOUBLE]
    else:
        b = None
        recvbuf_b = None

    comm.Gatherv([b_part, int(M_part), MPI.DOUBLE], recvbuf_b, root=0)
    
    # Запись результата
    if rank == 0:
        elapsed_time = time.time() - start_time
        print(f"Execution time: {elapsed_time:.4f} seconds")

        f4 = open('Results_parallel_scatterv.dat', 'w')
```

## Вопросы для защиты

### Теоретические вопросы

**1. Основы MPI**
1. Что такое MPI и для чего он используется?
MPI (Message Passing Interface) - это стандарт для организации параллельных вычислений в системах с распределенной памятью. В таких системах каждый процессор имеет свою локальную память, и для обмена данными между процессами используется передача сообщений.

2. Какие основные компоненты MPI вы знаете?
comm – группа процессов, которые участвуют в вычислениях
MPI.COMM_WORLD – все процессы, на которых запущена программа
numprocs – количество процессов, участвующих в вычислениях
rank – номер (ранг) данного процесса
MPI.Init() - начало параллельных участков программы
MPI.Finalize() - конец параллельных участков программы
Send и Recv - обеспечивают взаимодействие ровно двух процессов
BCast - вовлекают все процессы коммуникатора и выполняются одновременно
Probe - оптимизация сбора информации с помощью функции 
Gatherv, Scatterv - операции коллективного взаимодействия
Send, Bsend, Ssend и Rsend - функции передачи сообщений

3. Что такое коммуникатор? Какие коммуникаторы бывают?
Коммуникатор - служебный объект, который объединяет группу процессов и контекст для выполнения операций обмена. Он создаёт логическую область связи.
Виды коммуникаторов:
1) Интра-коммуникаторы (intracommunicators) - для операций внутри группы процессов.
2) Интер-коммуникаторы (intercommunicators) - для обмена данными между двумя разными группами процессов.

4. Что такое ранг процесса и для чего он нужен?
Ранг процесса - уникальный идентификатор, который присваивается каждому процессу внутри определённого коммуникатора. Ранг позволяет процессам взаимодействовать и выполнять разные части программы в зависимости от своего номера.

**2. Модель передачи сообщений**
5. В чем разница между операциями точка-точка и коллективными операциями?
Разница между операциями точка-точка и коллективными в параллельном программировании (например, в MPI) заключается в количестве участвующих процессов, способе взаимодействия и назначении.
В операциях точка-точка участвуют ровно два процесса: один выступает в роли отправителя, другой — получателя. В коллективных операциях одновременно участвуют все процессы, входящие в заданный коммуникатор. Каждый процесс должен вызвать соответствующую коллективную функцию, возможно, с собственным набором параметров.

6. Что такое синхронная и асинхронная передача сообщений?
При синхронной передаче процесс-отправитель блокируется в функции отправки до тех пор, пока процесс-получатель не подтвердит, что сообщение получено и начало обрабатываться. То есть вызов функции отправки не возвращает управление, пока не завершится операция приёма на стороне получателя.
При асинхронной передаче функция отправки или приёма возвращает управление сразу после инициализации операции, не дожидаясь её фактического завершения. Процесс продолжает выполняться дальше, а передача данных происходит параллельно в фоновом режиме. 

7. Что такое дедлок (deadlock) и как его избежать?
Дедлок (deadlock) - ситуация взаимной блокировки, когда два или более процессов бесконечно ожидают события, которое никогда не произойдёт.
Как его избежать?
1) Для каждой операции отправки MPI_Send должна существовать парная ей операция приёма MPI_Recv, причём они должны быть синхронизированы во времени.
2) Неблокирующие операции (MPI_Isend, MPI_Irecv) позволяют процессу продолжить выполнение, не дожидаясь завершения операции отправки. Это даёт возможность начать приём сообщения до того, как будет выполнена отправка, и тем самым избежать взаимной блокировки.
3) Функция Recv - атомарная операция, которая одновременно отправляет и принимает данные, используя два буфера. Это исключает возможность дедлока, так как операция не может завершиться, пока не выполнится приём.

8. Какие варианты функции Send вы знаете (Send, Bsend, Ssend, Rsend)?
Bsend – посылка сообщений с буферизацией, то есть предварительно выделяется место в памяти (не буфер MPI), куда будет сразу скопирован массив. Тогда процесс будет сразу продолжать работу, не дожидаясь вызова Recv. Из минусов: необходимо выделять память дополнительно, т.е. памяти может не хватить; также Recv уже может быть вызван, но сообщение все равно будет копироваться.
Ssend – посылка с синхронизацией, выход из функции осуществляется после фактической отправки сообщения.
Rsend – посылка сообщения по готовности, сообщение сразу же уходит соответствующему процессу, независимо от того, был ли уже на нём вызван Recv. Если Recv ещё не был вызван, будет сбой.

**3. Коллективные операции**
9. Для чего нужны функции `Bcast`, `Scatterv` и `Gatherv`?
BCast рассылает одни и те же данные от одного процесса (корневого, заданного параметром root) всем остальным процессам в группе
Scatterv разбивает единый массив из корневого процесса на фрагменты и рассылает их разным процессам группы
Gatherv собирает данные от всех процессов и собирает их в одном корневом процессе.

10. В чем отличие `Scatter` от `Scatterv`?
Scatter требует, чтобы данные в буфере корневого процесса были непрерывными (расположены в памяти подряд), а порции, которые отправляются процессам - равными по размеру. Scatterv не имеет подобных ограничений.

11. В чем отличие `Gather` от `Gatherv`?
Gather предполагает, что все процессы отправляют данные одинакового размера. Корневой процесс получает сообщения и размещает их в своём буфере приёма в порядке возрастания рангов отправителей. Gatherv не имеет подобных ограничений.

12. Почему коллективные операции эффективнее операций точка-точка?
Коллективные операции вызываются на всех процессах коммуникатора, в то время как операции точка-точка вызываются каждая на конкретном процессе.

13. Что такое `Allgatherv` и для чего он нужен?
Allgatherv - функция, предназначенная для параллельных вычислений в системах с распределённой памятью. Её ключевая особенность в том, что она позволяет собирать данные от всех процессов группы и рассылать их обратно всем участникам одновременно, при этом каждый процесс может передавать данные разного объёма.

**4. Распределение данных**
14. Что такое `rcounts` и `displs`? Для чего они нужны?
Массив rcounts содержит количество элементов, которые каждый процесс отправляет в точку сбора (например, в корневой процесс). Индекс в этом массиве соответствует рангу отправляющего процесса. Он задаёт "размер" сообщения от каждого процесса. Это особенно полезно, когда разные процессы генерируют данные разного объёма, и функция передачи должна корректно собрать их в единый буфер в точке сбора.
Массив displs определяет смещения (в элементах) относительно начала буфера приёма в точке сбора. Индекс в этом массиве также соответствует рангу отправляющего процесса. Он указывает, с какого смещения в буфере корня начинаются данные, полученные от каждого процесса. Благодаря этому корневому процессу не нужно копировать информацию из разрозненных буверов отдельных процессов - он просто размещает их в нужных местах своего выходного буфера.

15. Как обработать случай, когда `M` не делится на число процессов?
1) вычисляем базовое количество элементов на процесс и остаток;
2) формируем массив размеров блоков;
3) распределяем данные с помощью scatterv;
4) выполняем вычисления на локальных данных;
5) собираем результаты

16. Какие стратегии распределения данных вы знаете?
1) Блочное распределение. Данные (например, массив или строки матрицы) разбиваются на непрерывные блоки, и каждый процесс получает свой блок. Это обеспечивает локальность: процесс работает с компактным, непрерывным участком памяти, что отлично для кэш-эффективности и многих матричных операций. Минус: если размер данных не делится нацело на число процессов, последний процесс может получить меньше элементов, что ведёт к дисбалансу. 
2) Циклическое распределение. Элементы (или итерации цикла) распределяются по процессам по кругу: каждый процесс обрабатывает элементы с индексами, начиная со своего номера, с шагом, равным числу процессов. Такой подход хорошо балансирует нагрузку — даже если в какой-то момент часть процессов простаивает, они быстро получат работу. Но есть нюанс: при таком распределении соседние элементы, нужные для вычислений, часто оказываются на разных процессах, что усложняет доступ к данным и может снижать локальность. 
3) Блочно-циклическое распределение. Это компромисс: данные сначала разбиваются на крупные блоки, а затем эти блоки распределяются между процессами циклически. Можно настраивать размер блоков — чем они меньше, тем выше гибкость балансировки, но может расти overhead из-за большего числа обменов. 
4) Динамическое распределение. Применяется, когда характер задачи меняется в процессе выполнения (например, в методах частиц или адаптивных сетках). После каждой фазы вычислений система анализирует загрузку и при необходимости перераспределяет данные с более занятых процессов на менее загруженные. Это сложнее в реализации, но позволяет эффективно использовать ресурсы на протяжении всего жизненного цикла задачи.

17. Как организовано хранение матрицы в вашей программе?
В файле in.dat хранятся два числа – N и M – на разных строках. В файлах AData.da и xData.dat хранятся построчно значения A и x соответственно.

```bash
import time

start_time = time.time()
```

19. Что такое ускорение (speedup) и эффективность (efficiency)?
Ускорение показывает, во сколько раз параллельная версия программы выполняется быстрее по сравнению с её последовательной версией, запущенной на одном процессоре.
Эффективность показывает, насколько рационально используются вычислительные ресурсы при параллельном выполнении.

20. Почему ускорение не бывает идеальным (равным числу процессов)?
1) В любой программе есть участки кода, которые по своей природе выполняются строго последовательно. 
2) Процессы часто вынуждены ждать друг друга в точках синхронизации (например, при барьерах). 
3) Если нагрузка распределена неравномерно (один процесс выполняет намного больше работы, чем другой), то при увеличении числа процессов часть из них будет дольше выполнять свою долю задачи, чем другие.

21. Какие факторы ограничивают масштабируемость?
1) Рост объёма обмена данными
2) Синхронизация процессов
3) Неравномерное распределение работы
4) Проблемы декомпозиции данных

22. Что такое закон Амдала и как он влияет на ускорение?
Закон гласит, что максимальное ускорение, которое можно получить, ограничено той частью программы, которая не может быть распараллелена. Если доля последовательных вычислений велика, то добавление процессоров даёт всё меньший и меньший выигрыш.

**7. Сравнение версий**
27. В чем преимущества и недостатки каждой версии?
1) При последовательном выполнении программа запускается как единый последовательный процесс. Все вычисления выполняются одним процессом, без распределения задач между несколькими процессорами.
Преимущества: простота отладки и тестирования, инимальное время на разработку, низкие накладные расходы.
Недостатки: низкая производительность на многопроцессорных системах, неэффективное использование ресурсов, ограниченная масштабируемость.
2) При параллельном выполнении через send/recv задача распределяется между несколькими процессами, которые обмениваются данными с помощью базовых примитивов MPI - Send (отправитель) и Recv (получатель). 
Преимущества: гибкость, контроль, масштабируемость.
Недостатки: сложность реализации, накладные расходы на передачу, риск ошибок синхронизации.
3) При параллельном выполнении через scatterv данные распределяются от одного процесса (корневого) по всем остальным процессам группы. Каждый процесс получает свою порцию данных. 
Преимущества: простота программирования, структурированность, эффективное использование механизмов MPI.
Недостатки: ограниченная гибкость, жёсткая схема распределения, риск дисбаланса нагрузки.

28. Какая версия работает быстрее и почему?
Scatterv обычно быстрее, чем явные Send/Recv в MPI, а последовательная версия — самая медленная. вместо numproc − 1 отдельных Send/Recv, делается один вызов Scatter. Это сильно снижает накладные расходы; операция спроектирована так, чтобы быть эффективной на уровне всей группы процессов; нет риска deadlock, код чище и легче отлаживается.

45. Как повлияет на производительность использование `Sendrecv`?
Если процесс одновременно и отправляет, и принимает данные (например, обменивается «пограничными» значениями в задачах вычислительной геометрии или обработки изображений), Sendrecv позволяет сделать это за один системный вызов. Это устраняет лишнюю синхронизацию и накладные расходы на переключение между двумя отдельными вызовами (send + recv), что особенно заметно в циклах с частым обменом.
Если обмен данными строго однонаправленный (один процесс отправляет, другой только принимает), использование Sendrecv вместо раздельных вызовов может добавить ненужную абстракцию. В таких случаях раздельные вызовы иногда оказываются эффективнее из-за более тонкой настройки под конкретную задачу.

46. Что такое `MPI.Reduce_scatter` и для чего он нужен?
MPI_Reduce_scatter - коллективная функция, которая совмещает две операции: редукцию (объединение данных) и рассылку результатов по процессам группы.
MPI_Reduce_scatter удобна в сценариях, где нужно сначала собрать и обработать данные со всех процессов (например, посчитать сумму всех элементов или найти максимальное значение), а затем распределить итоговые результаты по разным частям вычислений для дальнейшей обработки. Это позволяет избежать явного разделения данных на этапе передачи и упрощает логику программы.

50. Что такое сильное и слабое масштабирование?
При сильном масштабировании размер задачи остаётся фиксированным, а количество процессов увеличивается. Цель - добиться линейного роста ускорения (то есть сокращения времени выполнения) с ростом числа процессов.
Слабое масштабирование предполагает, что одновременно с ростом числа процессоров увеличивается и размер задачи. Ключевой принцип - время выполнения алгоритма остаётся постоянным. Таким образом, объём работы, приходящийся на один процессор, не меняется.

**13. Архитектура**
51. Как влияет топология сети на производительность MPI-программ?
Если в системе есть прямые, "глубокие" связи между узлами, это часто даёт выигрыш: меньше промежуточных узлов, меньше задержек при передаче сообщений. В то же время для некоторых задач такая избыточность не нужна - проще организовать обмен в виде регулярной сетки, где каждый процесс общается с соседями.
Многие операции MPI реализуются через последовательность обменов. Алгоритм их выполнения напрямую зависит от топологии. Например, в плотной топологии (где много связей между далёкими узлами) алгоритм может задействовать больше ресурсов, чем в разреженной.

52. Что такое NUMA и как это влияет на производительность?
NUMA (Non-Uniform Memory Access) - архитектура организации памяти в многопроцессорных системах, при которой время доступа к памяти зависит от её физического расположения относительно процессора. Процессор быстрее получает данные из своей локальной памяти (подключённой напрямую к его контроллеру), а доступ к удалённой памяти (подключённой к другим процессорам или их группам) занимает больше времени, так как запрос проходит через межпроцессорную шину.
Как NUMA влияет на производительность:
1) Минимизация задержек. Если приложение работает с данными, которые находятся в локальной памяти процессора, доступ к ним происходит быстро, что снижает общую задержку и повышает пропускную способность.
2) Снижение нагрузки на межпроцессорную шину. Когда приложение использует локальную память, трафик через шину между узлами снижается, что положительно сказывается на пропускной способности системы.
3) Высокие задержки при доступе к удалённой памяти. Если процесс, выполняющийся на одном процессоре, часто обращается к данным на другом процессоре (в другом NUMA-узле), каждый такой запрос проходит через межпроцессорную шину, что значительно замедляет работу. Разница в латентности (времени доступа) может составлять 30–50% и более.
4) Перегрузка межпроцессорной шины. Если множество потоков постоянно читают данные из чужой (удалённой) памяти, шина между процессорами может быть перегружена, что замедляет работу всей системы.
5) Миграция процессов. Если операционная система или планировщик MPI-процессов перемещает процессы между разными NUMA-узлами, это может привести к тому, что потоки одного процесса начнут работать с данными на разных узлах, что увеличивает количество медленных удалённых обращений.

53. Как использовать привязку процессов к ядрам процессора?
При привязке явно задаётся, на каких конкретных логических ядрах процессора будут выполняться ваши MPI-процессы. Это особенно важно на многоядерных системах, где миграция процесса на другое ядро может привести к повторной загрузке данных из медленной памяти (L3-кэш, удаленная память).

54. Что такое гибридное программирование (MPI+OpenMP)?
Гибридное программирование MPI+OpenMP - подход к параллельным вычислениям, при котором программа использует обе технологии - MPI и OpenMP - для эффективного использования ресурсов современных высокопроизводительных вычислительных систем. Такие системы обычно состоят из многоядерных узлов с общей памятью (SMP-кластеров), объединённых сетью.
